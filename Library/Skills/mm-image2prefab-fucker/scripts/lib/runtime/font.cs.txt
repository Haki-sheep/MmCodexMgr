using System;
using System.Linq;
using UnityEngine;
using UnityEngine.TextCore.LowLevel;
using UnityEditor;
using TMPro;

public static class DxxUiFont
{
    [Serializable] public class Config { public string source; public string target; public string characters; public bool dryRun; }

    /// <summary>
    /// 从已有字体源新建独立SDF资源 预热本页文案 不覆盖共享字体
    /// </summary>
    public static object Execute()
    {
        if (EditorApplication.isPlayingOrWillChangePlaymode || EditorApplication.isCompiling || EditorApplication.isUpdating) throw new InvalidOperationException("需要空闲Edit Mode");
        var config = JsonUtility.FromJson<Config>(System.Text.Encoding.UTF8.GetString(Convert.FromBase64String("__PAYLOAD_BASE64__")));
        if (!config.dryRun && AssetDatabase.LoadMainAssetAtPath(config.target) != null) throw new InvalidOperationException("拒绝覆盖字体 " + config.target);
        var source = AssetDatabase.LoadAssetAtPath<Font>(config.source);
        if (source == null) throw new InvalidOperationException("源字体不存在 " + config.source);
        // TMP_Settings 随 Essential Resources 导入 字体资产创建依赖动态字体配置
        if (TMP_Settings.instance == null) throw new InvalidOperationException("缺少 TMP Essential Resources 请先导入 Window/TextMeshPro/Import TMP Essential Resources");
        if (FontEngine.LoadFontFace(source, 90) != FontEngineError.Success) throw new InvalidOperationException("源字体加载失败 " + config.source);
        var face = FontEngine.GetFaceInfo();
        var font = TMP_FontAsset.CreateFontAsset(source, 90, 9, GlyphRenderMode.SDFAA, 1024, 1024, AtlasPopulationMode.Dynamic, true);
        if (font == null) throw new InvalidOperationException("TMP 字体资产创建失败 " + config.source);
        bool persisted = false;
        try
        {
            string missing;
            if (!font.TryAddCharacters(config.characters, out missing)) throw new InvalidOperationException("字体缺字 " + missing);
            font.name = System.IO.Path.GetFileNameWithoutExtension(config.target);
            if (config.dryRun) return new { passed = true, dry_run = true, source = config.source, face.familyName, face.styleName, glyphs = font.characterTable.Count, changed_assets = new string[0] };
            string current = "Assets";
            foreach (string part in config.target.Substring(0, config.target.LastIndexOf('/')).Split('/').Skip(1))
            {
                string next = current + "/" + part;
                if (!AssetDatabase.IsValidFolder(next)) AssetDatabase.CreateFolder(current, part);
                current = next;
            }
            AssetDatabase.CreateAsset(font, config.target);
            persisted = true;
            int count = 0;
            foreach (var atlas in font.atlasTextures)
            {
                if (atlas == null) continue;
                atlas.name = font.name + " Atlas " + count++;
                AssetDatabase.AddObjectToAsset(atlas, font);
            }
            font.material.name = font.name + " Material";
            AssetDatabase.AddObjectToAsset(font.material, font);
            EditorUtility.SetDirty(font);
            AssetDatabase.SaveAssetIfDirty(font);
            return new { passed = true, path = config.target, source = config.source, face.familyName, face.styleName,
                glyphs = font.characterTable.Count, atlases = count, changed_assets = new[] { config.target } };
        }
        finally
        {
            if (!persisted)
            {
                foreach (var atlas in font.atlasTextures) if (atlas != null) UnityEngine.Object.DestroyImmediate(atlas);
                if (font.material != null) UnityEngine.Object.DestroyImmediate(font.material);
                UnityEngine.Object.DestroyImmediate(font);
            }
        }
    }
}
