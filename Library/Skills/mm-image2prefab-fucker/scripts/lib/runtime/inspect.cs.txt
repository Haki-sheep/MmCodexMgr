using System;
using System.Linq;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.Rendering;
using UnityEngine.SceneManagement;
using UnityEngine.EventSystems;
using UnityEditor;
using UnityEditor.SceneManagement;
using TMPro;
public static class UiInspectTemplate {
 [Serializable] public class Config {public string prefab,output;public int width,height;public bool render;public float toleranceX=.5f,toleranceY=.5f;}
 static string PathOf(Transform t,Transform root){return t==root?root.name:PathOf(t.parent,root)+"/"+t.name;}
 public static object Execute() {
  if(EditorApplication.isPlayingOrWillChangePlaymode||EditorApplication.isCompiling||EditorApplication.isUpdating)throw new InvalidOperationException("Idle Edit Mode required");
  var cfg=JsonUtility.FromJson<Config>(System.Text.Encoding.UTF8.GetString(Convert.FromBase64String("__PAYLOAD_BASE64__")));
  var prefab=AssetDatabase.LoadAssetAtPath<GameObject>(cfg.prefab);if(prefab==null)throw new InvalidOperationException("Prefab missing");
  var errors=new List<string>();var warnings=new List<string>();var textRows=new List<object>();var links=new List<object>();var layoutRows=new List<object>();
  var glyphs=new DxxUiFontGlyphs();
  foreach(var t in prefab.GetComponentsInChildren<Transform>(true)) {
   string path=PathOf(t,prefab.transform);
   if(GameObjectUtility.GetMonoBehavioursWithMissingScriptCount(t.gameObject)>0)errors.Add("Missing script: "+path);
   var r=t as RectTransform;
   if(r!=null&&(float.IsNaN(r.rect.width)||float.IsInfinity(r.rect.width)||float.IsNaN(r.rect.height)||float.IsInfinity(r.rect.height)))errors.Add("Nonfinite rect: "+path);
   if(PrefabUtility.IsAnyPrefabInstanceRoot(t.gameObject))links.Add(new{node=path,source=PrefabUtility.GetPrefabAssetPathOfNearestInstanceRoot(t.gameObject)});
   if(t.GetComponent<LayoutGroup>()!=null&&t.GetComponent<ContentSizeFitter>()!=null)warnings.Add("Review layout ownership: "+path);
   var layout=t.GetComponent<LayoutGroup>();if(layout!=null)layoutRows.Add(new{node=path,type=layout.GetType().Name,rect=r==null?null:new{width=r.rect.width,height=r.rect.height}});
  }
  foreach(var t in prefab.GetComponentsInChildren<TMP_Text>(true)) {
   string path=PathOf(t.transform,prefab.transform);var missing=new List<string>();
   if(t.font==null)errors.Add("Missing TMP font: "+path);
   else for(int i=0;i<t.text.Length;i++){int cp=char.ConvertToUtf32(t.text,i);if(char.IsHighSurrogate(t.text[i]))i++;if(!char.IsWhiteSpace((char)cp)&&!glyphs.HasCharacter(t.font,cp))missing.Add("U+"+cp.ToString("X"));}
   if(missing.Count>0)warnings.Add("Glyphs not yet in font atlas: "+path+" "+string.Join(",",missing.Distinct().ToArray()));
   if(t.raycastTarget)warnings.Add("TMP receives raycasts: "+path);
   textRows.Add(new{node=path,text=t.text,font=t.font==null?null:AssetDatabase.GetAssetPath(t.font),color=new[]{t.color.r,t.color.g,t.color.b,t.color.a},font_size=t.fontSize,alignment=t.alignment.ToString(),auto_size=t.enableAutoSizing,missing_glyphs=missing.Distinct().ToArray()});
  }
  foreach(var b in prefab.GetComponentsInChildren<Button>(true)) {
   if(b.targetGraphic==null)errors.Add("Button targetGraphic missing: "+b.name);
   else if(!b.GetComponentsInChildren<Graphic>(true).Any(g=>g.raycastTarget))errors.Add("Button has no raycast graphic: "+b.name);
  }
  string rendered=null;string renderState="not_run";var geometry=new List<string>();var hitRows=new List<object>();
  var active=SceneManager.GetActiveScene();bool dirty=active.isDirty;
  if(cfg.render) {
   // Only instantiate UI with no custom behaviours: ExecuteAlways callbacks are not a sandbox.
   var custom=prefab.GetComponentsInChildren<MonoBehaviour>(true).Where(b=>b!=null&&b.GetType().Assembly!=typeof(Image).Assembly&&b.GetType().Assembly!=typeof(TMP_Text).Assembly&&b.GetType().Assembly!=typeof(EventSystem).Assembly).Select(b=>b.GetType().FullName).Distinct().ToArray();
   if(custom.Length>0){renderState="blocked_custom_behaviours";warnings.Add("Use project scenario or read-only current GameView capture for custom behaviours: "+string.Join(",",custom));}
   else {
    var scene=EditorSceneManager.NewPreviewScene();RenderTexture rt=null;Texture2D tex=null;var oldRt=RenderTexture.active;var previousEventSystem=EventSystem.current;
    try {
     var canvasGO=new GameObject("UiTaskPreview",typeof(RectTransform),typeof(Canvas),typeof(GraphicRaycaster));SceneManager.MoveGameObjectToScene(canvasGO,scene);canvasGO.layer=5;
     var canvas=canvasGO.GetComponent<Canvas>();canvas.renderMode=RenderMode.WorldSpace;var cr=(RectTransform)canvasGO.transform;cr.sizeDelta=new Vector2(cfg.width,cfg.height);cr.localScale=Vector3.one*.01f;
     var go=(GameObject)PrefabUtility.InstantiatePrefab(prefab,scene);go.transform.SetParent(cr,false);
     foreach(var t in go.GetComponentsInChildren<Transform>(true))t.gameObject.layer=5;
     foreach(var cv in go.GetComponentsInChildren<Canvas>(true))cv.enabled=false;
     foreach(var scaler in go.GetComponentsInChildren<CanvasScaler>(true))scaler.enabled=false;
     foreach(var es in go.GetComponentsInChildren<EventSystem>(true))es.enabled=false;
     var r=go.transform as RectTransform;if(r==null)throw new InvalidOperationException("UI root requires RectTransform");r.anchorMin=Vector2.zero;r.anchorMax=Vector2.one;r.offsetMin=r.offsetMax=Vector2.zero;r.localScale=Vector3.one;
     var cg=new GameObject("UiTaskCamera",typeof(Camera));SceneManager.MoveGameObjectToScene(cg,scene);var camera=cg.GetComponent<Camera>();camera.scene=scene;camera.transform.position=new Vector3(0,0,-30);camera.orthographic=true;camera.orthographicSize=cfg.height*.005f;camera.aspect=(float)cfg.width/cfg.height;camera.cullingMask=1<<5;camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.clear;camera.allowHDR=false;canvas.worldCamera=camera;
     rt=new RenderTexture(cfg.width,cfg.height,24,RenderTextureFormat.ARGB32);rt.Create();camera.targetTexture=rt;
     Canvas.ForceUpdateCanvases();LayoutRebuilder.ForceRebuildLayoutImmediate(r);foreach(var t in go.GetComponentsInChildren<TMP_Text>())t.ForceMeshUpdate(true,true);Canvas.ForceUpdateCanvases();
     foreach(var t in go.GetComponentsInChildren<TMP_Text>()) {
      var preferred=t.GetPreferredValues();var rect=t.rectTransform.rect;
      if(t.isTextTruncated)geometry.Add("Text truncated: "+PathOf(t.transform,go.transform));
      else if(t.isTextOverflowing||preferred.y>rect.height+.5f||!t.enableWordWrapping&&preferred.x>rect.width+.5f)warnings.Add("Review text bounds visually: "+PathOf(t.transform,go.transform));
     }
     if(GraphicsSettings.currentRenderPipeline==null)camera.Render();
     else {var request=new RenderPipeline.StandardRequest{destination=rt};if(!RenderPipeline.SupportsRenderRequest(camera,request))throw new InvalidOperationException("Pipeline does not support StandardRequest");RenderPipeline.SubmitRenderRequest(camera,request);}
     // 以渲染后的实际四角检查画布范围 不只检查尺寸是否为正
     foreach(var box in go.GetComponentsInChildren<RectTransform>()) {
      string nodePath=PathOf(box,go.transform);
      if(box.rect.width<=0||box.rect.height<=0)geometry.Add("Nonpositive rect: "+nodePath);
      var corners=new Vector3[4];box.GetWorldCorners(corners);
      var points=corners.Select(point=>r.InverseTransformPoint(point)).ToArray();
      if(points.Any(point=>point.x<r.rect.xMin-cfg.toleranceX||point.x>r.rect.xMax+cfg.toleranceX||point.y<r.rect.yMin-cfg.toleranceY||point.y>r.rect.yMax+cfg.toleranceY))geometry.Add("Outside canvas: "+nodePath);
      else if(points.Any(point=>point.x<r.rect.xMin-.5f||point.x>r.rect.xMax+.5f||point.y<r.rect.yMin-.5f||point.y>r.rect.yMax+.5f))warnings.Add("Minor canvas overflow within reference tolerance: "+nodePath);
     }
     // Hit target inspection only; never invoke gameplay callbacks here.
     var eg=new GameObject("UiTaskEventSystem",typeof(EventSystem));SceneManager.MoveGameObjectToScene(eg,scene);var eventSystem=eg.GetComponent<EventSystem>();eventSystem.enabled=false;
     foreach(var b in go.GetComponentsInChildren<Button>()) {
      var pos=(Vector2)camera.WorldToScreenPoint(((RectTransform)b.transform).TransformPoint(((RectTransform)b.transform).rect.center));
      var ev=new PointerEventData(eventSystem){position=pos};var hits=new List<RaycastResult>();canvasGO.GetComponent<GraphicRaycaster>().Raycast(ev,hits);
      var target=hits.Count==0?null:ExecuteEvents.GetEventHandler<IPointerClickHandler>(hits[0].gameObject);
      hitRows.Add(new{node=PathOf(b.transform,go.transform),correct_target=target==b.gameObject,interactable=b.IsInteractable()});
     }
     RenderTexture.active=rt;tex=new Texture2D(cfg.width,cfg.height,TextureFormat.RGBA32,false);tex.ReadPixels(new Rect(0,0,cfg.width,cfg.height),0,0);tex.Apply();
     System.IO.File.WriteAllBytes(cfg.output,tex.EncodeToPNG());rendered=cfg.output;renderState="captured_requires_visual_review";
    } catch(Exception e){renderState="failed";errors.Add("Render: "+e.Message);}
    finally {RenderTexture.active=oldRt;if(tex!=null)UnityEngine.Object.DestroyImmediate(tex);if(rt!=null){rt.Release();UnityEngine.Object.DestroyImmediate(rt);}EditorSceneManager.ClosePreviewScene(scene);if(previousEventSystem!=null)EventSystem.current=previousEventSystem;}
   }
  }
  return new{mode="online_fallback",changed_assets=new string[0],structural_passed=errors.Count==0,errors=errors.ToArray(),warnings=warnings.ToArray(),texts=textRows.ToArray(),layouts=layoutRows.ToArray(),nested_instances=links.ToArray(),geometry=geometry.ToArray(),hit_targets=hitRows.ToArray(),render_state=renderState,screenshot=rendered,capture_scope="isolated UI camera; not final GameView",visual_validation="requires_reference_review",layout_validation=renderState=="captured_requires_visual_review"?(geometry.Count==0?"passed":"failed"):"not_run",business_validation="not_run",scene_unchanged=active==SceneManager.GetActiveScene()&&dirty==active.isDirty};
 }
}
