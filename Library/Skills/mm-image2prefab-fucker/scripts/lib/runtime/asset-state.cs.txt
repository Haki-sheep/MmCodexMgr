public static class DxxUiText
{
    public static void Apply(TMPro.TMP_Text text, DxxUiModel.Node node)
    {
        text.fontSize = node.fontSize;
        text.color = new UnityEngine.Color(node.color[0], node.color[1], node.color[2], node.color[3]);
        text.enableAutoSizing = node.autoSize;
        if (node.autoSize) { text.fontSizeMin = 1; text.fontSizeMax = node.fontSize; }
    }
}

public static class DxxUiModel
{
    [System.Serializable] public class Input { public string action; public string mode; public Plan plan; public AssetState baseline; public string[] removedIds; }
    [System.Serializable] public class Plan { public string prefabPath; public string adaptation; public Node[] nodes; public Import[] imports; public ComponentPlan[] components; }
    [System.Serializable] public class ComponentPlan { public string id; public Plan plan; }
    [System.Serializable] public class LayoutConfig
    {
        public string type; public string constraint; public int constraintCount; public float[] cellSize; public float[] spacing; public int[] padding;
        public string childAlignment; public string startCorner; public string startAxis;
        public bool childControlWidth; public bool childControlHeight; public bool childForceExpandWidth; public bool childForceExpandHeight;
    }
    [System.Serializable] public class LayoutElement { public float[] minSize; public float[] preferredSize; public float[] flexibleSize; public bool ignoreLayout; }
    [System.Serializable] public class Import { public string source; public string destination; public string sha256; public bool reuse; }
    [System.Serializable] public class RectData { public float[] anchorMin; public float[] anchorMax; public float[] pivot; public float[] position; public float[] size; }
    [System.Serializable] public class Node
    {
        public string id; public string name; public string parent; public string kind;
        public RectData rect; public string sprite; public string text; public string font;
        public float fontSize; public string prefab; public bool active; public bool sliced; public bool preserveAspect;
        public int siblingIndex; public float[] color; public string componentId;
        public string fontStyle; public string alignment; public bool wordWrap; public string overflow; public bool autoSize;
        public Override[] overrides; public LayoutConfig layout; public LayoutElement layoutElement; public string[] fields;
    }
    [System.Serializable] public class Override
    {
        public string targetId; public int[] path; public string text; public string font; public float fontSize;
        public string sprite; public float[] color; public bool active; public string[] fields;
    }
    [System.Serializable] public class Binding
    {
        public string id; public string gameObjectFileId; public string rectTransformFileId; public string textFileId; public string imageFileId; public string layoutGroupFileId; public string layoutElementFileId;
    }
    [System.Serializable] public class Dependency { public string path; public string guid; public string sha256; }
    [System.Serializable] public class AssetState
    {
        public string path; public string guid; public string sha256; public Binding[] nodeBindings; public Dependency[] dependencies;
    }
}

public static class DxxUiAssetState
{
    /// <summary>
    /// 保存后读取真实资产身份及节点file ID 供后续原位更新核对
    /// </summary>
    public static DxxUiModel.AssetState Capture(string path, System.Collections.Generic.Dictionary<string, int[]> paths)
    {
        var root = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.GameObject>(path);
        if (root == null) throw new System.InvalidOperationException("已保存Prefab无法重新读取 " + path);
        var bindings = new System.Collections.Generic.List<DxxUiModel.Binding>();
        foreach (var pair in paths)
        {
            var target = Resolve(root.transform, pair.Value).gameObject;
            bindings.Add(new DxxUiModel.Binding { id = pair.Key, gameObjectFileId = FileId(target),
                rectTransformFileId = FileId(target.GetComponent<UnityEngine.RectTransform>()),
                textFileId = FileId(target.GetComponent<TMPro.TMP_Text>()), imageFileId = FileId(target.GetComponent<UnityEngine.UI.Image>()),
                layoutGroupFileId = FileId(target.GetComponent<UnityEngine.UI.LayoutGroup>()), layoutElementFileId = FileId(target.GetComponent<UnityEngine.UI.LayoutElement>()) });
        }
        var dependencies = new System.Collections.Generic.List<DxxUiModel.Dependency>();
        foreach (string dependency in UnityEditor.AssetDatabase.GetDependencies(path, true))
        {
            if (dependency == path || !dependency.StartsWith("Assets/") || !System.IO.File.Exists(dependency)) continue;
            dependencies.Add(new DxxUiModel.Dependency { path = dependency, guid = UnityEditor.AssetDatabase.AssetPathToGUID(dependency), sha256 = Hash(dependency) });
        }
        return new DxxUiModel.AssetState { path = path, guid = UnityEditor.AssetDatabase.AssetPathToGUID(path), sha256 = Hash(path), nodeBindings = bindings.ToArray(), dependencies = dependencies.ToArray() };
    }

    public static string Hash(string path)
    {
        using (var sha = System.Security.Cryptography.SHA256.Create()) return System.BitConverter.ToString(sha.ComputeHash(System.IO.File.ReadAllBytes(path))).Replace("-", "").ToLowerInvariant();
    }

    public static string FileId(UnityEngine.Object value)
    {
        if (value == null) return "0";
        string guid; long id;
        if (!UnityEditor.AssetDatabase.TryGetGUIDAndLocalFileIdentifier(value, out guid, out id)) throw new System.InvalidOperationException("对象没有持久资产身份 " + value.name);
        return id.ToString(System.Globalization.CultureInfo.InvariantCulture);
    }

    public static UnityEngine.Transform Resolve(UnityEngine.Transform root, int[] path)
    {
        foreach (int index in path)
        {
            if (index < 0 || index >= root.childCount) throw new System.InvalidOperationException("实例覆盖子节点索引不存在");
            root = root.GetChild(index);
        }
        return root;
    }

    public static int[] PathOf(UnityEngine.Transform node, UnityEngine.Transform root)
    {
        var path = new System.Collections.Generic.List<int>();
        while (node != root) { if (node.parent == null) throw new System.InvalidOperationException("节点不在Prefab内"); path.Add(node.GetSiblingIndex()); node = node.parent; }
        path.Reverse(); return path.ToArray();
    }

    /// <summary>
    /// 核对原Prefab身份 允许用户移动及撤销后继续原位更新
    /// </summary>
    public static void Check(DxxUiModel.AssetState baseline)
    {
        if (baseline == null || string.IsNullOrEmpty(baseline.guid) || UnityEditor.AssetDatabase.AssetPathToGUID(baseline.path) != baseline.guid) throw new System.InvalidOperationException("原Prefab资产身份已变化 请核对目标路径");
    }

    public static int[] TemplatePath(DxxUiModel.Plan plan, string id)
    {
        var path = new System.Collections.Generic.List<int>();
        while (id != "root")
        {
            var node = System.Linq.Enumerable.Single(plan.nodes, item => item.id == id);
            path.Add(node.siblingIndex); id = node.parent;
        }
        path.Reverse(); return path.ToArray();
    }

    /// <summary>
    /// 仅覆盖既有实例的文本图标与显示状态 不改模板或业务事件
    /// </summary>
    public static void ApplyOverrides(UnityEngine.GameObject instance, DxxUiModel.Node node, DxxUiModel.Plan plan)
    {
        foreach (var patch in node.overrides ?? new DxxUiModel.Override[0])
        {
            var template = string.IsNullOrEmpty(node.componentId) ? null : System.Linq.Enumerable.Single(plan.components, item => item.id == node.componentId).plan;
            var target = Resolve(instance.transform, template == null ? patch.path : TemplatePath(template, patch.targetId)).gameObject;
            var fields = patch.fields ?? new string[0];
            var text = target.GetComponent<TMPro.TMP_Text>(); var image = target.GetComponent<UnityEngine.UI.Image>();
            if (System.Linq.Enumerable.Any(fields, field => field == "text" || field == "font" || field == "fontSize") && text == null) throw new System.InvalidOperationException("文字覆盖目标缺少TMP " + node.id);
            if (System.Linq.Enumerable.Contains(fields, "sprite") && image == null) throw new System.InvalidOperationException("图标覆盖目标缺少Image " + node.id);
            if (System.Linq.Enumerable.Contains(fields, "text")) text.text = patch.text;
            if (System.Linq.Enumerable.Contains(fields, "font")) { text.font = UnityEditor.AssetDatabase.LoadAssetAtPath<TMPro.TMP_FontAsset>(patch.font); if (text.font == null) throw new System.InvalidOperationException("覆盖字体不存在 " + patch.font); }
            if (System.Linq.Enumerable.Contains(fields, "text") || System.Linq.Enumerable.Contains(fields, "font")) new DxxUiFontGlyphs().RequireText(text.font, text.text, node.id + " override");
            if (System.Linq.Enumerable.Contains(fields, "fontSize")) text.fontSize = patch.fontSize;
            if (System.Linq.Enumerable.Contains(fields, "sprite")) { image.sprite = UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.Sprite>(patch.sprite); if (image.sprite == null) throw new System.InvalidOperationException("覆盖Sprite不存在 " + patch.sprite); }
            if (System.Linq.Enumerable.Contains(fields, "color")) { var graphic = target.GetComponent<UnityEngine.UI.Graphic>(); if (graphic == null) throw new System.InvalidOperationException("颜色覆盖目标缺少Graphic"); graphic.color = new UnityEngine.Color(patch.color[0], patch.color[1], patch.color[2], patch.color[3]); }
            if (System.Linq.Enumerable.Contains(fields, "active")) target.SetActive(patch.active);
            if (UnityEditor.PrefabUtility.IsPartOfPrefabInstance(target))
            {
                UnityEditor.PrefabUtility.RecordPrefabInstancePropertyModifications(target);
                foreach (var component in target.GetComponents<UnityEngine.Component>()) UnityEditor.PrefabUtility.RecordPrefabInstancePropertyModifications(component);
            }
            UnityEditor.EditorUtility.SetDirty(target);
            foreach (var component in target.GetComponents<UnityEngine.Component>()) UnityEditor.EditorUtility.SetDirty(component);
        }
    }
}
