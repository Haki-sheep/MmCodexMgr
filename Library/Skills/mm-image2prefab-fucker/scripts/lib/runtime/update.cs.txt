using System;
using System.Linq;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using UnityEditor;
using UnityEditor.SceneManagement;
using TMPro;
using static DxxUiModel;
using UiLayoutElement = UnityEngine.UI.LayoutElement;

public static class DxxUiUpdate
{
    private static TextAnchor Alignment(string value) { return (TextAnchor)Enum.Parse(typeof(TextAnchor), value); }

    private static void ApplyLayout(GameObject gameObject, Node node)
    {
        if (node.fields.Contains("layout"))
        {
            var previous = gameObject.GetComponent<LayoutGroup>();
            bool same = previous != null && node.layout != null && ((node.layout.type == "grid" && previous is GridLayoutGroup) || (node.layout.type == "horizontal" && previous is HorizontalLayoutGroup) || (node.layout.type == "vertical" && previous is VerticalLayoutGroup));
            if (previous != null && !same) UnityEngine.Object.DestroyImmediate(previous);
        }
        if (node.fields.Contains("layout") && node.layout != null && node.layout.type != "overlay")
        {
            LayoutGroup group = gameObject.GetComponent<LayoutGroup>();
            if (group == null) group = node.layout.type == "grid" ? (LayoutGroup)gameObject.AddComponent<GridLayoutGroup>() : node.layout.type == "horizontal" ? (LayoutGroup)gameObject.AddComponent<HorizontalLayoutGroup>() : (LayoutGroup)gameObject.AddComponent<VerticalLayoutGroup>();
            group.padding = new RectOffset(node.layout.padding[0], node.layout.padding[2], node.layout.padding[1], node.layout.padding[3]); group.childAlignment = Alignment(node.layout.childAlignment);
            if (group is GridLayoutGroup grid)
            {
                grid.cellSize = Vector(node.layout.cellSize); grid.spacing = Vector(node.layout.spacing); grid.startCorner = (GridLayoutGroup.Corner)Enum.Parse(typeof(GridLayoutGroup.Corner), node.layout.startCorner); grid.startAxis = (GridLayoutGroup.Axis)Enum.Parse(typeof(GridLayoutGroup.Axis), node.layout.startAxis); grid.constraint = node.layout.constraint == "fixedRowCount" ? GridLayoutGroup.Constraint.FixedRowCount : node.layout.constraint == "flexible" ? GridLayoutGroup.Constraint.Flexible : GridLayoutGroup.Constraint.FixedColumnCount; grid.constraintCount = node.layout.constraintCount;
            }
            else if (group is HorizontalLayoutGroup horizontal) { horizontal.spacing = node.layout.spacing[0]; horizontal.childControlWidth = node.layout.childControlWidth; horizontal.childControlHeight = node.layout.childControlHeight; horizontal.childForceExpandWidth = node.layout.childForceExpandWidth; horizontal.childForceExpandHeight = node.layout.childForceExpandHeight; }
            else if (group is VerticalLayoutGroup vertical) { vertical.spacing = node.layout.spacing[1]; vertical.childControlWidth = node.layout.childControlWidth; vertical.childControlHeight = node.layout.childControlHeight; vertical.childForceExpandWidth = node.layout.childForceExpandWidth; vertical.childForceExpandHeight = node.layout.childForceExpandHeight; }
        }
        if (node.fields.Contains("layoutElement") && node.layoutElement == null && gameObject.GetComponent<UiLayoutElement>() != null) UnityEngine.Object.DestroyImmediate(gameObject.GetComponent<UiLayoutElement>());
        if (node.fields.Contains("layoutElement") && node.layoutElement != null)
        {
            var element = gameObject.GetComponent<UiLayoutElement>();
            if (element == null) element = gameObject.AddComponent<UiLayoutElement>();
            element.ignoreLayout = node.layoutElement.ignoreLayout; element.minWidth = node.layoutElement.minSize[0]; element.minHeight = node.layoutElement.minSize[1]; element.preferredWidth = node.layoutElement.preferredSize[0]; element.preferredHeight = node.layoutElement.preferredSize[1]; element.flexibleWidth = node.layoutElement.flexibleSize[0]; element.flexibleHeight = node.layoutElement.flexibleSize[1];
        }
    }

    private static void RebuildLayouts(GameObject root)
    {
        Canvas.ForceUpdateCanvases();
        foreach (var group in root.GetComponentsInChildren<LayoutGroup>(true)) LayoutRebuilder.ForceRebuildLayoutImmediate(group.transform as RectTransform);
        Canvas.ForceUpdateCanvases();
    }

    /// <summary>
    /// 按稳定绑定修改指定字段及结构 保留既有对象和持久事件引用
    /// </summary>
    public static object Execute()
    {
        if (EditorApplication.isPlayingOrWillChangePlaymode || EditorApplication.isCompiling || EditorApplication.isUpdating || PrefabStageUtility.GetCurrentPrefabStage() != null) throw new InvalidOperationException("需要空闲Edit Mode并退出Prefab Stage");
        var input = __INPUT__;
        DxxUiAssetState.Check(input.baseline);
        if (input.action == "folders")
        {
            foreach (var import in input.plan.imports.Where(item => !item.reuse))
            {
                string current = "Assets";
                foreach (string part in import.destination.Substring(0, import.destination.LastIndexOf('/')).Split('/').Skip(1))
                { string next = current + "/" + part; if (!AssetDatabase.IsValidFolder(next)) AssetDatabase.CreateFolder(current, part); current = next; }
            }
            return new { passed = true };
        }
        if (input.action == "imports")
        {
            foreach (var import in input.plan.imports.Where(item => !item.reuse))
            {
                var importer = AssetImporter.GetAtPath(import.destination) as TextureImporter;
                if (importer == null) throw new InvalidOperationException("缺少纹理导入器 " + import.destination);
                importer.textureType = TextureImporterType.Sprite; importer.spriteImportMode = SpriteImportMode.Single;
                importer.alphaIsTransparency = true; importer.textureCompression = TextureImporterCompression.Uncompressed; importer.SaveAndReimport();
            }
            return new { passed = true };
        }
        var asset = AssetDatabase.LoadAssetAtPath<GameObject>(input.plan.prefabPath);
        var sourceObjects = asset.GetComponentsInChildren<Transform>(true).ToDictionary(item => DxxUiAssetState.FileId(item.gameObject), item => item.gameObject);
        var paths = new Dictionary<string, int[]>();
        var removed = new HashSet<string>(input.removedIds ?? new string[0]);
        var baselineIds = new HashSet<string>(input.baseline.nodeBindings.Select(item => item.id));
        foreach (var binding in input.baseline.nodeBindings)
        {
            GameObject target;
            if (removed.Contains(binding.id) && !sourceObjects.ContainsKey(binding.gameObjectFileId)) continue;
            if (!sourceObjects.TryGetValue(binding.gameObjectFileId, out target) || DxxUiAssetState.FileId(target.GetComponent<RectTransform>()) != binding.rectTransformFileId
                || DxxUiAssetState.FileId(target.GetComponent<TMP_Text>()) != binding.textFileId || DxxUiAssetState.FileId(target.GetComponent<Image>()) != binding.imageFileId
                || (binding.layoutGroupFileId != null && DxxUiAssetState.FileId(target.GetComponent<LayoutGroup>()) != binding.layoutGroupFileId)
                || (binding.layoutElementFileId != null && DxxUiAssetState.FileId(target.GetComponent<UiLayoutElement>()) != binding.layoutElementFileId)) throw new InvalidOperationException("目标节点已删除或组件身份已变化 无法继续原位定位 " + binding.id);
            paths.Add(binding.id, DxxUiAssetState.PathOf(target.transform, asset.transform));
        }
        foreach (var node in input.plan.nodes)
        {
            if (!paths.ContainsKey(node.id) && baselineIds.Contains(node.id)) throw new InvalidOperationException("缺少节点绑定 " + node.id);
            if (!paths.ContainsKey(node.id) && !node.fields.Contains("create")) throw new InvalidOperationException("新增节点缺少明确创建指令 " + node.id);
            if (node.kind == "text") new DxxUiFontGlyphs().RequireText(AssetDatabase.LoadAssetAtPath<TMP_FontAsset>(node.font), node.text, node.id + " " + node.font);
            if (!string.IsNullOrEmpty(node.sprite) && AssetDatabase.LoadAssetAtPath<Sprite>(node.sprite) == null) throw new InvalidOperationException("Sprite不存在 " + node.sprite);
            if (node.kind == "prefab" && !paths.ContainsKey(node.id) && AssetDatabase.LoadAssetAtPath<GameObject>(node.prefab) == null) throw new InvalidOperationException("组件Prefab不存在 " + node.prefab);
        }
        var sourceTargets = paths.ToDictionary(pair => pair.Key, pair => DxxUiAssetState.Resolve(asset.transform, pair.Value));
        var managed = new HashSet<Transform>(sourceTargets.Values);
        foreach (string id in removed.Where(sourceTargets.ContainsKey))
            foreach (Transform child in sourceTargets[id].GetComponentsInChildren<Transform>(true))
                if (!managed.Contains(child) && !(sourceTargets.Values.Any(parent => PrefabUtility.IsAnyPrefabInstanceRoot(parent.gameObject) && child.IsChildOf(parent)) && PrefabUtility.GetCorrespondingObjectFromSource(child.gameObject) != null)) throw new InvalidOperationException("删除容器包含未登记的用户对象 请明确处理 " + id + " " + child.name);
        if (input.action == "validate") return new { passed = true, changed_assets = new string[0] };
        var root = PrefabUtility.LoadPrefabContents(input.plan.prefabPath);
        try
        {
            var targets = paths.ToDictionary(pair => pair.Key, pair => DxxUiAssetState.Resolve(root.transform, pair.Value).gameObject);
            foreach (var node in input.plan.nodes)
            {
                bool created = !targets.ContainsKey(node.id);
                if (created)
                {
                    var gameObject = node.kind == "prefab" ? (GameObject)PrefabUtility.InstantiatePrefab(AssetDatabase.LoadAssetAtPath<GameObject>(node.prefab), root.scene) : new GameObject(node.name, typeof(RectTransform));
                    gameObject.layer = 5;
                    if (node.kind == "image" || node.kind == "button") { var image = gameObject.AddComponent<Image>(); image.raycastTarget = node.kind == "button"; if (node.kind == "button") gameObject.AddComponent<Button>().targetGraphic = image; }
                    if (node.kind == "text") { var text = gameObject.AddComponent<TextMeshProUGUI>(); text.raycastTarget = false; }
                    targets.Add(node.id, gameObject);
                    node.fields = new string[] { "name", "parent", "rect", "text", "font", "fontSize", "color", "autoSize", "fontStyle", "alignment", "wordWrap", "overflow", "sprite", "preserveAspect", "overrides", "layout", "layoutElement", "active", "siblingIndex" };
                }
                var target = targets[node.id];
                if (node.fields.Contains("name")) target.name = node.name;
                if (node.fields.Contains("parent") && node.id != "root") target.transform.SetParent(targets[node.parent].transform, false);
                var rect = target.GetComponent<RectTransform>();
                if (node.fields.Contains("rect"))
                {
                    rect.anchorMin = Vector(node.rect.anchorMin); rect.anchorMax = Vector(node.rect.anchorMax); rect.pivot = Vector(node.rect.pivot);
                    rect.sizeDelta = Vector(node.rect.size); rect.anchoredPosition = Vector(node.rect.position);
                    if (created) { rect.localScale = Vector3.one; rect.localRotation = Quaternion.identity; }
                }
                if (node.kind == "text")
                {
                    var text = target.GetComponent<TMP_Text>();
                    if (node.fields.Contains("text")) text.text = node.text;
                    if (node.fields.Contains("font")) text.font = AssetDatabase.LoadAssetAtPath<TMP_FontAsset>(node.font);
                    if (node.fields.Contains("fontSize")) text.fontSize = node.fontSize;
                    if (node.fields.Contains("color")) text.color = ColorOf(node.color);
                    if (node.fields.Contains("autoSize")) text.enableAutoSizing = node.autoSize;
                    if (node.autoSize && (node.fields.Contains("autoSize") || node.fields.Contains("fontSize"))) { text.fontSizeMin = 1; text.fontSizeMax = node.fontSize; }
                    if (node.fields.Contains("fontStyle")) text.fontStyle = node.fontStyle == "BoldItalic" ? FontStyles.Bold | FontStyles.Italic : (FontStyles)Enum.Parse(typeof(FontStyles), node.fontStyle);
                    if (node.fields.Contains("alignment")) text.alignment = (TextAlignmentOptions)Enum.Parse(typeof(TextAlignmentOptions), node.alignment);
                    if (node.fields.Contains("wordWrap")) text.enableWordWrapping = node.wordWrap;
                    if (node.fields.Contains("overflow")) text.overflowMode = (TextOverflowModes)Enum.Parse(typeof(TextOverflowModes), node.overflow);
                }
                if (node.kind == "image" || node.kind == "button")
                {
                    var image = target.GetComponent<Image>();
                    if (node.fields.Contains("color")) image.color = ColorOf(node.color);
                    if (node.fields.Contains("sprite")) image.sprite = string.IsNullOrEmpty(node.sprite) ? null : AssetDatabase.LoadAssetAtPath<Sprite>(node.sprite);
                    if (node.fields.Contains("preserveAspect")) image.preserveAspect = node.preserveAspect;
                    if (created && node.sliced) { if (image.sprite == null || image.sprite.border == Vector4.zero) throw new InvalidOperationException("九宫格需要已配置border的Sprite " + node.id); image.type = Image.Type.Sliced; }
                }
                if (node.kind == "prefab" && node.fields.Contains("overrides")) DxxUiAssetState.ApplyOverrides(target, node, input.plan);
                ApplyLayout(target, node);
                if (node.fields.Contains("active")) target.SetActive(node.active);
                if (PrefabUtility.IsPartOfPrefabInstance(target))
                {
                    PrefabUtility.RecordPrefabInstancePropertyModifications(target);
                    foreach (var component in target.GetComponents<Component>()) PrefabUtility.RecordPrefabInstancePropertyModifications(component);
                }
            }
            foreach (string id in removed.Where(targets.ContainsKey)) { if (targets[id] != null) UnityEngine.Object.DestroyImmediate(targets[id]); targets.Remove(id); }
            // 先按原路径绑定全部对象 再调整同父级绘制顺序 避免索引位移误改其他节点
            foreach (var node in input.plan.nodes.Where(node => node.fields.Contains("siblingIndex")).OrderBy(node => node.siblingIndex)) targets[node.id].transform.SetSiblingIndex(node.siblingIndex);
            RebuildLayouts(root.transform.gameObject);
            paths.Clear();
            foreach (var pair in targets) paths[pair.Key] = DxxUiAssetState.PathOf(pair.Value.transform, root.transform);
            DxxUiAssetState.Check(input.baseline);
            bool success; PrefabUtility.SaveAsPrefabAsset(root, input.plan.prefabPath, out success);
            if (!success) throw new InvalidOperationException("Prefab原位保存失败");
        }
        finally { PrefabUtility.UnloadPrefabContents(root); }
        var state = DxxUiAssetState.Capture(input.plan.prefabPath, paths);
        var identityChanges = state.nodeBindings.Where(binding => {
            var before = input.baseline.nodeBindings.FirstOrDefault(item => item.id == binding.id);
            if (before == null) return false;
            var node = input.plan.nodes.FirstOrDefault(item => item.id == binding.id);
            return before.gameObjectFileId != binding.gameObjectFileId || before.rectTransformFileId != binding.rectTransformFileId || before.textFileId != binding.textFileId || before.imageFileId != binding.imageFileId
                || (node != null && !node.fields.Contains("layout") && before.layoutGroupFileId != null && before.layoutGroupFileId != binding.layoutGroupFileId)
                || (before.layoutGroupFileId == "0" && (node == null || node.layout == null) && binding.layoutGroupFileId != "0")
                || (node != null && node.layout != null && node.layout.type != "overlay" && binding.layoutGroupFileId == "0")
                || (node != null && !node.fields.Contains("layoutElement") && before.layoutElementFileId != null && before.layoutElementFileId != binding.layoutElementFileId)
                || (before.layoutElementFileId == "0" && (node == null || node.layoutElement == null) && binding.layoutElementFileId != "0")
                || (node != null && node.layoutElement != null && binding.layoutElementFileId == "0");
        }).Select(binding => { var before = input.baseline.nodeBindings.First(item => item.id == binding.id); return binding.id + " before=" + JsonUtility.ToJson(before) + " after=" + JsonUtility.ToJson(binding); }).ToArray();
        if (state.guid != input.baseline.guid || identityChanges.Length != 0) throw new InvalidOperationException("原位保存后对象身份变化 保留现场禁止重放 " + string.Join(" ", identityChanges));
        return new { passed = true, prefab = input.plan.prefabPath, asset_state = state };
    }

    private static Vector2 Vector(float[] value) { return new Vector2(value[0], value[1]); }
    private static Color ColorOf(float[] value) { return new Color(value[0], value[1], value[2], value[3]); }
}
