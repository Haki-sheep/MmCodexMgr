using UnityEditor.SceneManagement;

public static class Image2PrefabEditorStage
{
    /// <summary>
    /// 读取预制体编辑状态 保持当前编辑现场
    /// </summary>
    public static object Execute()
    {
        var Stage = PrefabStageUtility.GetCurrentPrefabStage();
        return new { prefabStageOpen = Stage != null };
    }
}
