using System;
using System.Collections.Generic;
using System.Linq;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.SceneManagement;
using UnityEditor;
using UnityEditor.SceneManagement;
using TMPro;
using static DxxUiModel;
using UiLayoutElement = UnityEngine.UI.LayoutElement;

public static class DxxUiAssembly
{
    private static Vector2 Vector(float[] value) { return new Vector2(value[0], value[1]); }
    private static Color ColorOf(float[] value) { return value == null ? Color.white : new Color(value[0], value[1], value[2], value[3]); }

    private static TextAnchor Alignment(string value) { return (TextAnchor)Enum.Parse(typeof(TextAnchor), value); }

    private static void ApplyLayout(GameObject gameObject, Node node)
    {
        if (node.layout != null && node.layout.type != "overlay")
        {
            LayoutGroup group = node.layout.type == "grid" ? (LayoutGroup)gameObject.AddComponent<GridLayoutGroup>() : node.layout.type == "horizontal" ? (LayoutGroup)gameObject.AddComponent<HorizontalLayoutGroup>() : (LayoutGroup)gameObject.AddComponent<VerticalLayoutGroup>();
            group.padding = new RectOffset(node.layout.padding[0], node.layout.padding[2], node.layout.padding[1], node.layout.padding[3]);
            group.childAlignment = Alignment(node.layout.childAlignment);
            if (group is GridLayoutGroup grid)
            {
                grid.cellSize = Vector(node.layout.cellSize); grid.spacing = Vector(node.layout.spacing);
                grid.startCorner = (GridLayoutGroup.Corner)Enum.Parse(typeof(GridLayoutGroup.Corner), node.layout.startCorner);
                grid.startAxis = (GridLayoutGroup.Axis)Enum.Parse(typeof(GridLayoutGroup.Axis), node.layout.startAxis);
                grid.constraint = node.layout.constraint == "fixedRowCount" ? GridLayoutGroup.Constraint.FixedRowCount : node.layout.constraint == "flexible" ? GridLayoutGroup.Constraint.Flexible : GridLayoutGroup.Constraint.FixedColumnCount;
                grid.constraintCount = node.layout.constraintCount;
            }
            else if (group is HorizontalLayoutGroup horizontal)
            {
                horizontal.spacing = node.layout.spacing[0]; horizontal.childControlWidth = node.layout.childControlWidth; horizontal.childControlHeight = node.layout.childControlHeight; horizontal.childForceExpandWidth = node.layout.childForceExpandWidth; horizontal.childForceExpandHeight = node.layout.childForceExpandHeight;
            }
            else if (group is VerticalLayoutGroup vertical)
            {
                vertical.spacing = node.layout.spacing[1]; vertical.childControlWidth = node.layout.childControlWidth; vertical.childControlHeight = node.layout.childControlHeight; vertical.childForceExpandWidth = node.layout.childForceExpandWidth; vertical.childForceExpandHeight = node.layout.childForceExpandHeight;
            }
        }
        if (node.layoutElement != null)
        {
            var element = gameObject.AddComponent<UiLayoutElement>();
            element.ignoreLayout = node.layoutElement.ignoreLayout; element.minWidth = node.layoutElement.minSize[0]; element.minHeight = node.layoutElement.minSize[1]; element.preferredWidth = node.layoutElement.preferredSize[0]; element.preferredHeight = node.layoutElement.preferredSize[1]; element.flexibleWidth = node.layoutElement.flexibleSize[0]; element.flexibleHeight = node.layoutElement.flexibleSize[1];
        }
    }

    private static void RebuildLayouts(GameObject root)
    {
        Canvas.ForceUpdateCanvases();
        foreach (var group in root.GetComponentsInChildren<LayoutGroup>(true)) LayoutRebuilder.ForceRebuildLayoutImmediate(group.transform as RectTransform);
        Canvas.ForceUpdateCanvases();
    }

    private static void CreateFolder(string folder)
    {
        string current = "Assets";
        foreach (string part in folder.Split('/').Skip(1))
        {
            string next = current + "/" + part;
            if (!AssetDatabase.IsValidFolder(next)) AssetDatabase.CreateFolder(current, part);
            current = next;
        }
    }

    private static T Load<T>(string file) where T : UnityEngine.Object
    {
        var asset = AssetDatabase.LoadAssetAtPath<T>(file);
        if (asset == null) throw new InvalidOperationException("缺少资源 " + file);
        return asset;
    }

    private static void CheckFont(Node node)
    {
        var font = Load<TMP_FontAsset>(node.font);
        var glyphs = new DxxUiFontGlyphs();
        glyphs.RequireText(font, node.text, node.name + " " + node.font);
    }

    /// <summary>
    /// 仅新建经计划的 UI 资产 不覆盖既有 Prefab 不接线业务事件
    /// </summary>
    public static object Execute()
    {
        if (EditorApplication.isPlayingOrWillChangePlaymode || EditorApplication.isCompiling || EditorApplication.isUpdating) throw new InvalidOperationException("需要空闲 Edit Mode");
        if (PrefabStageUtility.GetCurrentPrefabStage() != null) throw new InvalidOperationException("请先退出当前 Prefab Stage");
        var input = __INPUT__;
        var plan = input.plan;
        if (System.IO.File.Exists(plan.prefabPath)) throw new InvalidOperationException("拒绝覆盖已有 Prefab 请原位更新");
        foreach (var node in plan.nodes)
        {
            if (node.kind == "text") CheckFont(node);
            if (node.kind == "prefab")
            {
                if (!string.IsNullOrEmpty(node.componentId) && input.action != "assemble") continue;
                var prefab = Load<GameObject>(node.prefab);
                var custom = prefab.GetComponentsInChildren<MonoBehaviour>(true).Where(item => item != null && item.GetType().Assembly != typeof(Image).Assembly && item.GetType().Assembly != typeof(TMP_Text).Assembly && item.GetType().Assembly != typeof(UnityEngine.EventSystems.EventSystem).Assembly).ToArray();
                if (custom.Length != 0) throw new InvalidOperationException("通用装配拒绝含自定义生命周期脚本的组件 " + node.prefab);
            }
        }
        foreach (var image in plan.imports.Where(item => item.reuse)) Load<Sprite>(image.destination);
        if (input.action == "validate") return new { passed = true, nodes = plan.nodes.Length, changed_assets = new string[0] };
        if (input.action == "folders")
        {
            CreateFolder(plan.prefabPath.Substring(0, plan.prefabPath.LastIndexOf('/')));
            foreach (var image in plan.imports) CreateFolder(image.destination.Substring(0, image.destination.LastIndexOf('/')));
            return new { passed = true };
        }
        foreach (var image in plan.imports)
        {
            if (image.reuse) continue;
            var importer = AssetImporter.GetAtPath(image.destination) as TextureImporter;
            if (importer == null) throw new InvalidOperationException("缺少纹理导入器 " + image.destination);
            importer.textureType = TextureImporterType.Sprite;
            importer.spriteImportMode = SpriteImportMode.Single;
            importer.alphaIsTransparency = true;
            importer.textureCompression = TextureImporterCompression.Uncompressed;
            importer.SaveAndReimport();
        }
        var previous = SceneManager.GetActiveScene();
        bool dirty = previous.isDirty;
        var scene = EditorSceneManager.NewPreviewScene();
        var objects = new Dictionary<string, GameObject>();
        try
        {
            foreach (var node in plan.nodes)
            {
                var gameObject = node.kind == "prefab" ? (GameObject)PrefabUtility.InstantiatePrefab(Load<GameObject>(node.prefab), scene) : new GameObject(node.name, typeof(RectTransform));
                SceneManager.MoveGameObjectToScene(gameObject, scene);
                gameObject.name = node.name;
                gameObject.layer = 5;
                if (!string.IsNullOrEmpty(node.parent)) gameObject.transform.SetParent(objects[node.parent].transform, false);
                var rect = gameObject.transform as RectTransform;
                if (rect == null) throw new InvalidOperationException("组件根节点需要 RectTransform");
                rect.anchorMin = Vector(node.rect.anchorMin); rect.anchorMax = Vector(node.rect.anchorMax);
                rect.pivot = Vector(node.rect.pivot); rect.sizeDelta = Vector(node.rect.size);
                rect.anchoredPosition = Vector(node.rect.position); rect.localScale = Vector3.one;
                rect.localRotation = Quaternion.identity;
                if (node.kind == "image" || node.kind == "button")
                {
                    var image = gameObject.AddComponent<Image>();
                    if (!string.IsNullOrEmpty(node.sprite)) image.sprite = Load<Sprite>(node.sprite);
                    image.color = ColorOf(node.color); image.raycastTarget = node.kind == "button";
                    image.preserveAspect = node.preserveAspect;
                    if (node.sliced)
                    {
                        if (image.sprite == null || image.sprite.border == Vector4.zero) throw new InvalidOperationException("九宫格需要已配置 border 的 Sprite");
                        image.type = Image.Type.Sliced;
                    }
                    if (node.kind == "button") gameObject.AddComponent<Button>().targetGraphic = image;
                }
                if (node.kind == "text")
                {
                    var text = gameObject.AddComponent<TextMeshProUGUI>();
                    text.font = Load<TMP_FontAsset>(node.font); text.text = node.text;
                    DxxUiText.Apply(text, node);
                    text.fontStyle = node.fontStyle == "BoldItalic" ? FontStyles.Bold | FontStyles.Italic : string.IsNullOrEmpty(node.fontStyle) ? FontStyles.Normal : (FontStyles)Enum.Parse(typeof(FontStyles), node.fontStyle);
                    text.alignment = string.IsNullOrEmpty(node.alignment) ? TextAlignmentOptions.Center : (TextAlignmentOptions)Enum.Parse(typeof(TextAlignmentOptions), node.alignment); text.raycastTarget = false;
                    text.enableWordWrapping = node.wordWrap;
                    text.overflowMode = string.IsNullOrEmpty(node.overflow) ? TextOverflowModes.Overflow : (TextOverflowModes)Enum.Parse(typeof(TextOverflowModes), node.overflow);
                }
                gameObject.SetActive(node.active);
                objects.Add(node.id, gameObject);
            }
            foreach (var node in plan.nodes.Where(item => !string.IsNullOrEmpty(item.parent))) objects[node.id].transform.SetSiblingIndex(node.siblingIndex);
            foreach (var node in plan.nodes) ApplyLayout(objects[node.id], node);
            foreach (var node in plan.nodes.Where(item => item.kind == "prefab")) DxxUiAssetState.ApplyOverrides(objects[node.id], node, plan);
            RebuildLayouts(objects["root"]);
            foreach (var gameObject in objects.Values.Where(PrefabUtility.IsPartOfPrefabInstance))
            {
                PrefabUtility.RecordPrefabInstancePropertyModifications(gameObject);
                foreach (var component in gameObject.GetComponents<Component>()) PrefabUtility.RecordPrefabInstancePropertyModifications(component);
            }
            bool success;
            var paths = objects.ToDictionary(pair => pair.Key, pair => DxxUiAssetState.PathOf(pair.Value.transform, objects["root"].transform));
            var result = PrefabUtility.SaveAsPrefabAsset(objects["root"], plan.prefabPath, out success);
            if (!success || result == null) throw new InvalidOperationException("Prefab 保存失败");
            return new { passed = true, prefab = plan.prefabPath, asset_state = DxxUiAssetState.Capture(plan.prefabPath, paths), nodes = plan.nodes.Length, business_events = "not_bound", adaptation = plan.adaptation, scene_unchanged = SceneManager.GetActiveScene() == previous && previous.isDirty == dirty };
        }
        finally { EditorSceneManager.ClosePreviewScene(scene); }
    }
}
