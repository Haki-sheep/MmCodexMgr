using System;
using System.IO;
using System.Linq;
using System.Collections.Generic;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;

public static class MmEffectCapture
{
    [Serializable]
    public sealed class Job
    {
        /// <summary> 导入后的特效 JSON 资产路径 </summary>
        public string effect;
        /// <summary> 本次采样图输出目录 </summary>
        public string output;
        /// <summary> 秒为单位的采样时刻 </summary>
        public float[] timeList;
        /// <summary> Three.js 右手坐标系相机位置 </summary>
        public float[] positionList;
        /// <summary> Three.js 右手坐标系注视点 </summary>
        public float[] targetList;
        /// <summary> 视场角 </summary>
        public float fov;
        /// <summary> 图像宽度 </summary>
        public int width;
        /// <summary> 图像高度 </summary>
        public int height;
        /// <summary> 背景 HTML 色值 </summary>
        public string background;
        /// <summary> 仅持久化本次首次导入的效果材质 </summary>
        public bool saveImportedMaterials;
    }

    /// <summary>
    /// 在独立预览场景采样原生粒子并回传图片及粒子数
    /// </summary>
    public static object Execute()
    {
        var Config = JsonUtility.FromJson<Job>(@"__MM_EFFECT_JOB__");
        string PrefabPath = Config.effect.Replace(".fxforge.json", ".prefab");
        var Prefab = AssetDatabase.LoadAssetAtPath<GameObject>(PrefabPath);
        if (Prefab == null) throw new InvalidOperationException("Generated prefab missing Complete the structured import first " + PrefabPath);
        string EffectRoot = Path.GetDirectoryName(Path.GetDirectoryName(Config.effect)).Replace('\\', '/') + "/";
        var ActiveScene = SceneManager.GetActiveScene();
        var PreviewScene = EditorSceneManager.NewPreviewScene();
        var PreviousTarget = RenderTexture.active;
        var Texture = new RenderTexture(Config.width, Config.height, 24, RenderTextureFormat.ARGB32, RenderTextureReadWrite.sRGB);
        var ImageList = new List<string>();
        var CountList = new List<int[]>();
        var WarningList = new List<string>();
        Application.LogCallback Callback = (Message, Stack, Type) =>
        {
            if (Type == LogType.Warning || Type == LogType.Error || Type == LogType.Exception) WarningList.Add(Type + " " + Message);
        };
        Application.logMessageReceived += Callback;
        try
        {
            var Instance = (GameObject)PrefabUtility.InstantiatePrefab(Prefab, PreviewScene);
            var Root = Instance.GetComponent<ParticleSystem>();
            var SystemList = Instance.GetComponentsInChildren<ParticleSystem>(true).Where(System => System != Root).ToArray();
            var CameraObject = new GameObject("MmEffectCaptureCamera");
            SceneManager.MoveGameObjectToScene(CameraObject, PreviewScene);
            var Camera = CameraObject.AddComponent<Camera>();
            Camera.scene = PreviewScene;
            Camera.nearClipPlane = 0.05f;
            Camera.farClipPlane = 500;
            Camera.fieldOfView = Config.fov;
            Camera.clearFlags = CameraClearFlags.SolidColor;
            if (!ColorUtility.TryParseHtmlString(Config.background, out var Background)) throw new InvalidOperationException("Invalid background");
            Camera.backgroundColor = Background;
            Camera.allowHDR = true;
            Camera.depthTextureMode = DepthTextureMode.Depth;
            Camera.targetTexture = Texture;
            Camera.transform.position = new Vector3(Config.positionList[0], Config.positionList[1], -Config.positionList[2]);
            Camera.transform.LookAt(new Vector3(Config.targetList[0], Config.targetList[1], -Config.targetList[2]));
            Directory.CreateDirectory(Config.output);
            for (int I = 0; I < Config.timeList.Length; I++)
            {
                Root.Simulate(Mathf.Max(0.0001f, Config.timeList[I]), true, true, true);
                CountList.Add(SystemList.Select(System => System.particleCount).ToArray());
                Camera.Render();
                RenderTexture.active = Texture;
                var Image = new Texture2D(Config.width, Config.height, TextureFormat.RGB24, false);
                try
                {
                    Image.ReadPixels(new Rect(0, 0, Config.width, Config.height), 0, 0);
                    Image.Apply();
                    string ImagePath = Path.Combine(Config.output, "unity_" + I.ToString("D3") + ".png");
                    if (File.Exists(ImagePath)) throw new IOException("Image already exists " + ImagePath);
                    File.WriteAllBytes(ImagePath, Image.EncodeToPNG());
                    ImageList.Add(ImagePath);
                }
                finally { UnityEngine.Object.DestroyImmediate(Image); }
            }
            foreach (var Renderer in Instance.GetComponentsInChildren<ParticleSystemRenderer>(true))
                foreach (var Material in Renderer.sharedMaterials)
                    if (Material != null)
                    {
                        if (ShaderUtil.ShaderHasError(Material.shader)) throw new InvalidOperationException("Shader error " + Material.shader.name);
                        if (Config.saveImportedMaterials)
                        {
                            if (!AssetDatabase.GetAssetPath(Material).StartsWith(EffectRoot, StringComparison.Ordinal)) throw new InvalidOperationException("Material outside the imported effect " + Material.name);
                            AssetDatabase.SaveAssetIfDirty(Material);
                        }
                    }
            if (WarningList.Any(Message => Message.StartsWith("Error ") || Message.StartsWith("Exception "))) throw new InvalidOperationException(string.Join("\n", WarningList));
            return new { scope = "Editor ParticleSystem Simulate and Camera Render", unityVersion = Application.unityVersion, pipeline = UnityEngine.Rendering.GraphicsSettings.currentRenderPipeline == null ? "Built-in" : UnityEngine.Rendering.GraphicsSettings.currentRenderPipeline.GetType().Name, colorSpace = QualitySettings.activeColorSpace.ToString(), prefab = PrefabPath, layerList = SystemList.Select(System => System.name).ToArray(), timeList = Config.timeList, countList = CountList.ToArray(), imageList = ImageList.ToArray(), warningList = WarningList.Distinct().ToArray(), activeSceneUnchanged = SceneManager.GetActiveScene() == ActiveScene, playModeValidated = false };
        }
        finally
        {
            Application.logMessageReceived -= Callback;
            RenderTexture.active = PreviousTarget;
            Texture.Release();
            UnityEngine.Object.DestroyImmediate(Texture);
            EditorSceneManager.ClosePreviewScene(PreviewScene);
        }
    }
}
