using System;
using System.IO;
using MmPerformance;
using UnityEditor;
using UnityEngine;

[InitializeOnLoad]
public static class MmPerformanceValidation
{
    /// <summary> 验证任务专用 SessionState 前缀 </summary>
    private const string StatePrefix = "MmPerformanceValidation.";
    /// <summary> 初始采样请求 </summary>
    private const string CaptureId = "a1000000000000000000000000000001";
    /// <summary> 取消采样请求 </summary>
    private const string CancelId = "a1000000000000000000000000000002";
    /// <summary> 暂停采样请求 </summary>
    private const string PauseId = "a1000000000000000000000000000003";

    /// <summary>
    /// 注册跨域验证推进函数
    /// </summary>
    static MmPerformanceValidation() { EditorApplication.update += Tick; }

    /// <summary>
    /// 启动隔离工程验证 设置最长执行时间
    /// </summary>
    public static void Run()
    {
        SessionState.SetBool(StatePrefix + "active", true);
        SessionState.SetInt(StatePrefix + "phase", 0);
        SessionState.SetFloat(StatePrefix + "deadline", (float)EditorApplication.timeSinceStartup + 120);
    }

    /// <summary>
    /// 输入操作及参数 调用真实 Unity Provider
    /// </summary>
    private static Response Call(string operation, string arguments, string id)
    {
        var Request = new Invocation { contract_version = "1.0", provider_id = "mm-performance", request_id = id,
            operation = operation, payload = new Payload { arguments_json = arguments },
            editor = new EditorIdentity { project_path = Path.GetFullPath(Path.Combine(Application.dataPath, "..")),
                editor_instance_id = "isolated-native-validation", editor_session_id = "isolated-native-validation" } };
        return JsonUtility.FromJson<Response>(PerformanceProvider.Execute(JsonUtility.ToJson(Request)));
    }

    /// <summary>
    /// 输入断言条件 失败时抛出完整上下文
    /// </summary>
    private static void Check(bool condition, string message)
    {
        if (!condition) throw new InvalidOperationException(message);
    }

    /// <summary>
    /// 根据验证阶段推进 Play 采样 取消和暂停拒绝
    /// </summary>
    private static void Tick()
    {
        if (!SessionState.GetBool(StatePrefix + "active", false)) return;
        try
        {
            if (EditorApplication.timeSinceStartup > SessionState.GetFloat(StatePrefix + "deadline", 0)) throw new TimeoutException("Native validation timed out");
            if (EditorApplication.isCompiling || EditorApplication.isUpdating) return;
            int phase = SessionState.GetInt(StatePrefix + "phase", 0);
            switch (phase)
            {
                case 0:
                    string Normalized = ReportStore.NormalizeMissingNumbers("{\"previous\":NaN,\"literal\":\"NaN \\\"Infinity\\\"\",\"other\":-Infinity}");
                    Check(Normalized == "{\"previous\":null,\"literal\":\"NaN \\\"Infinity\\\"\",\"other\":null}", "JSON missing sentinel normalization failed");
                    Check(Call("probe", "{}", Guid.NewGuid().ToString("N")).success, "probe failed");
                    Check(!Call("start", "{}", Guid.NewGuid().ToString("N")).success, "Edit capture was not refused");
                    var Audit = Call("audit", "{}", "a1000000000000000000000000000004");
                    Check(Audit.success && Audit.data.state == "completed" && File.Exists(Audit.data.result_path), "audit failed " + Audit.error);
                    SessionState.SetInt(StatePrefix + "phase", 1);
                    EditorApplication.isPlaying = true;
                    break;
                case 1:
                    if (!EditorApplication.isPlaying || EditorApplication.isPlayingOrWillChangePlaymode && !EditorApplication.isPlaying) return;
                    var Started = Call("start", "{\"seconds\":1,\"warmup\":0.1,\"phase\":\"native-test\"}", CaptureId);
                    Check(Started.success && Started.data.state == "running", "start failed " + Started.error);
                    Check(Call("start", "{}", CaptureId).data.job_id == CaptureId, "stable request replay changed job");
                    Check(!Call("start", "{}", Guid.NewGuid().ToString("N")).success, "overlapping capture was not refused");
                    Check(Call("mark", "{\"job_id\":\"" + CaptureId + "\",\"text\":\"native-marker\"}", Guid.NewGuid().ToString("N")).success, "mark failed");
                    SessionState.SetInt(StatePrefix + "phase", 2);
                    break;
                case 2:
                    var Result = CaptureSession.Status(CaptureId);
                    if (Result.state == "running") return;
                    Check(Result.state == "completed" && Result.samples > 0, "capture incomplete " + Result.reason);
                    var Report = JsonUtility.FromJson<CaptureReport>(File.ReadAllText(Result.result_path));
                    Check(Report.samplesList.Count == Result.samples && Report.eventsList.Count == 1, "sample or event data missing");
                    Check(Call("start", "{\"seconds\":5,\"warmup\":0}", CancelId).success, "second start failed");
                    Check(Call("cancel", "{\"job_id\":\"" + CancelId + "\"}", Guid.NewGuid().ToString("N")).data.state == "cancelled", "cancel failed");
                    Check(Call("start", "{\"seconds\":5,\"warmup\":0}", PauseId).success, "pause start failed");
                    EditorApplication.isPaused = true;
                    SessionState.SetInt(StatePrefix + "phase", 3);
                    break;
                case 3:
                    var Paused = CaptureSession.Status(PauseId);
                    if (Paused.state == "running") return;
                    Check(Paused.state == "interrupted", "pause did not interrupt " + Paused.state);
                    Check(!Call("start", "{}", Guid.NewGuid().ToString("N")).success, "paused start was not refused");
                    EditorApplication.isPaused = false;
                    Complete(true, "Native probe audit capture stable-id overlap mark cancel pause checks passed");
                    break;
            }
        }
        catch (Exception Exception) { Complete(false, Exception.ToString()); }
    }

    /// <summary>
    /// 输入验证结果 保存机器可读证据并退出测试 Editor
    /// </summary>
    private static void Complete(bool success, string message)
    {
        SessionState.SetBool(StatePrefix + "active", false);
        string Output = Path.GetFullPath(Path.Combine(Application.dataPath, "../Validation/native-validation.json"));
        Directory.CreateDirectory(Path.GetDirectoryName(Output));
        File.WriteAllText(Output, JsonUtility.ToJson(new Validation { success = success, message = message, unity_version = Application.unityVersion }));
        EditorApplication.Exit(success ? 0 : 1);
    }

    [Serializable]
    public sealed class Response
    {
        /// <summary> Provider 操作成功 </summary>
        public bool success;
        /// <summary> 任务收据 </summary>
        public Job data;
        /// <summary> 失败上下文 </summary>
        public string error;
    }

    [Serializable]
    public sealed class Validation
    {
        /// <summary> 验证成功 </summary>
        public bool success;
        /// <summary> 验证说明 </summary>
        public string message;
        /// <summary> Unity 实际版本 </summary>
        public string unity_version;
    }
}
